//This outlines the portion of the C grammar that is allowed by bencth (specifically from C89/C90)
//Modeled after the syntax used in the c standard
//Terminals are enclosed in angle brackets
//Non-terminals are enclosed in double quotes
//Or is represented by a pipe ("|")
//<xxx>_opt is represented by a question mark ("?")

//A.1.3 Identifiers
<identifier> ::= <nondigit> | <identifier> <nondigit> | <identifier> <digit>

<nondigit> ::= "a-z" | "A-Z" | "_"

<digit> ::= "0-9"

//A.1.4 Constants
<constant> ::= <integer-constant>

<integer-constant> ::= <decimal-constant>

<decimal-constant> ::= <nonzero-digit> | <decimal-constant> <digit>

<nonzero-digit> ::= "1-9"

//A.2.1 Expressions
<primary-expression> ::= <identifier> | <constant> | "(" <expression> ")"

<unary-expression> ::= <primary-expression> | <unary-operator> <unary-expression>

<unary-operator> ::= "+" | "-" | "!" | "~" | "&"

<multiplicative-expression> ::= <unary-expression> | <multiplicative-expression> ("*" | "/" | "%") <unary-expression>

<additive-expression> ::= <multiplicative-expression> | <additive-expression> ("+" | "-") <multiplicative-expression>

<assignment-expression> ::= <additive-expression> | <unary-expression> <assignment-operator> <assignment-expression>

<assignment-operator> ::= "="

<expression> ::= <assignment-expression> | <expression> "," <assignment-expression>

//A.2.2 Declarations
<declaration> ::= <declaration-specifiers> <init-declarator-list>? ";"

<declaration-specifiers> ::= <type-specifier> <declaration-specifiers>?

//yes, this does mean that you have to put a space after the comma
//if you disagree,
//leave
<init-declarator-list> ::= <init-declarator> | <init-declarator> ", " <init-declarator-list>?

<init-declarator> ::= <declarator> | <declarator> "=" <initializer>

<type-specifier> ::= "int"

<declarator> ::= <direct-declarator>

<direct-declarator> ::= <identifier> | "(" <declarator> ")" | <direct-declarator> "(" <parameter-type-list> ")" | <direct-declarator> "(" <identifier-list>? ")"

<parameter-type-list> ::= <parameter-list> | <parameter-list> ", " "..."

<parameter-list> ::= <parameter-declaration> | <parameter-list> ", " <parameter-declaration>

<parameter-declaration> ::= <declaration-specifiers> <declarator>

<identifier-list> ::= <identifier> | <identifier-list> ", " <identifier>

<initializer> ::= <assignment-expression> | "{" <initializer-list> ", "? "}"

<initializer-list> ::= <initializer> | <initializer-list> "," <initializer>

//A.2.3 Statements
<statement> ::= <compound-statement> | <expression-statement> | <jump-statement>

<compound-statement> ::= "{" <declaration-list>? <statement-list>? "}"

<declaration-list> ::= <declaration> | <declaration-list> <declaration>

<statement-list> ::= <statement> | <statement-list> <statement>

<expression-statement> ::= <expression>? ";"

<jump-statement> ::= "return" <expression>? ";"

//A.2.4 External Definitions

<translation-unit> ::= <external-declaration>

<external-declaration> ::= <function-definition>

<function-definition> ::= <declaration-specifiers>? <declarator> <declaration-list>? <compound-statement>
